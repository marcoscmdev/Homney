## Convencion
PK  -> Clave primaria
FK  -> Foreign Key
NN  -> Not Null
UQ  -> Unique
AI  -> Auto incremental
U -> Unsigned
### USUARIO
id_usuario INT U AI PK
nombre VARCHAR(100) NN
email VARCHAR(150) UQ NN
clave VARCHAR(255) NN 
fecha_nacimiento DATE
avatar VARCHAR(255) 

### GASTO
id_gasto INT U AI PK
Concepto VARCHAR(255) NN 
Modo ENUM(efectivo, bancario)
Tipo ENUM (fijo, variable)NN
Importe DECIMAL (10,2) NN 
Categoria ENUM(alimentacion, vivienda, facturas, menaje, salud, ocio, transporte, otros)
Subcategoria VARCHAR(50)
Pendiente BOOLEAN NN
id_hogar INT  U FK

### TAREA
id_tarea INT U AI PK
nombre VARCHAR(150) NN
duracion TINYINT 
recurrencia TINYINT NN 
estado ENUM (pendiente, en_progreso, finalizada) NN
id_usuario INT U FK
id_hogar INT  FK

### HABITACION
id_habitacion INT U AI  PK
nombre VARCHAR(100)NN
tipo ENUM(cocina,generica,aseo,garaje,exterior)NN
id_hogar INT U  FK
### HOGAR
id_hogar INT U AI PK
clave_inv VARCHAR (255) U 
### MURO
id_muro INT U AI PK 
fecha_pub DATETIME NN
titulo VARCHAR (50) NN
cuerpo TEXT NN
id_hogar INT U  FK
### USUARIO_REALIZA_GASTO
id_usuario  INT U  FK
id_gasto INT U FK
fecha DATETIME 
PK (id_usuario, id_gasto)

### USUARIO_RESPONSABLE_HABITACION
id_usuario  INT U  FK
id_habitacion INT U FK
PK (id_usuario, id_habitacion)
### USUARIO_HOGAR
id_usuario INT U FK
id_hogar INT U FK
rol ENUM (admin, miembro) NN
PK (id_usuario, id_hogar)

### REPARTO_GASTO
id_usuario INT U FK
id_gasto INT U FK
importe DECIMAL(10,2)NN
pagador BOOLEAN NN
PK(id_usuario, id_gasto)

### TAREA_REALIZADA
id_tarea INT U FK
id_usuario INT U FK
fecha DATETIME NN
duracion_real TINYINT
PK(id_tarea, id_usuario, fecha)
